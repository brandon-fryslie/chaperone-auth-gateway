```go
var (
    globalRegistry = make(map[string]AuthStrategy)
    registryMu     sync.RWMutex
)

func Register(name string, strategy AuthStrategy) {
    registryMu.Lock()
    defer registryMu.Unlock()
    globalRegistry[name] = strategy
}

func Get(name string) (AuthStrategy, error) {
    registryMu.RLock()
    defer registryMu.RUnlock()

    strategy, ok := globalRegistry[name]
    if !ok {
        return nil, errors.ErrStrategyNotFound
    }
    return strategy, nil
}
```

**Request Cloning Pattern:**
```go
// internal/auth/clone.go (NEW)
func CloneRequest(req *http.Request) *http.Request {
    clone := req.Clone(req.Context())

    // Clone URL (shallow copy by req.Clone)
    if req.URL != nil {
        cloneURL := *req.URL
        clone.URL = &cloneURL
    }

    // Clone headers (deep copy)
    clone.Header = make(http.Header)
    for k, v := range req.Header {
        clone.Header[k] = append([]string(nil), v...)
    }

    // Body is NOT cloned (streaming, can't rewind)
    // Strategies must not modify body

    return clone
}
```

### Technical Notes

**Implementation Pattern:**
```go
// internal/auth/bearer.go (NEW)
package auth

import (
    "context"
    "fmt"
    "net/http"

    "github.com/bmf/chaperone/internal/errors"
    "github.com/bmf/chaperone/internal/log"
)

type BearerStrategy struct{}

func (s *BearerStrategy) Apply(ctx context.Context, req *http.Request, secret string) error {
    if secret == "" {
        return errors.ErrSecretNotFound
    }

    // Clone request to avoid mutation
    cloned := CloneRequest(req)

    // Set Authorization header
    cloned.Header.Set("Authorization", "Bearer "+secret)

    log.Debug(ctx, "injected bearer token authentication")

    *req = *cloned  // Copy back to original pointer
    return nil
}

func init() {
    Register("bearer", &BearerStrategy{})
}
```

**Security Considerations:**
- Secret NEVER logged (only log "injected bearer token authentication")
- Header set with Set() not Add() to prevent duplicate headers
- Empty secret rejected immediately

```go
import (
    "context"
    "fmt"
    "net/http"

    "github.com/bmf/chaperone/internal/errors"
    "github.com/bmf/chaperone/internal/log"

func (s *HeaderStrategy) Apply(ctx context.Context, req *http.Request, secret string) error {
```

```go
func (h *MITMHandler) forwardRequest(ctx context.Context, clientConn net.Conn, req *http.Request, hostname string, svc *service.Service) error {
    upstreamURL := fmt.Sprintf("https://%s%s", hostname, req.URL.RequestURI())
    upstreamReq, err := http.NewRequestWithContext(ctx, req.Method, upstreamURL, req.Body)
    if err != nil {
        return err
    }

    // Copy headers
    for k, v := range req.Header {
        upstreamReq.Header[k] = v
    }

    // >>> INJECT AUTHENTICATION (NEW) <<<

    // 1. Fetch secret
    secret, err := h.secretRegistry.Fetch(ctx, svc.CredentialRef)
    if err != nil {
        log.Error(ctx, "failed to fetch secret", err, "service", svc.HostPattern, "ref", svc.CredentialRef)
        h.sendErrorResponse(clientConn, http.StatusServiceUnavailable, "Secret unavailable")
        return err
    }
    log.Debug(ctx, "fetched secret", "provider", parseProvider(svc.CredentialRef))

    // 2. Get auth strategy
    strategy, err := h.authRegistry.Get(svc.AuthStrategyRef)
    if err != nil {
        log.Error(ctx, "unknown auth strategy", err, "service", svc.HostPattern, "strategy", svc.AuthStrategyRef)
        h.sendErrorResponse(clientConn, http.StatusBadGateway, "Authentication configuration error")
        return err
    }

    // 3. Apply authentication
    if err := strategy.Apply(ctx, upstreamReq, secret); err != nil {
        log.Error(ctx, "failed to apply authentication", err, "service", svc.HostPattern)
        h.sendErrorResponse(clientConn, http.StatusBadGateway, "Authentication failed")
        return err
    }
    log.Info(ctx, "applied authentication", "service", svc.HostPattern, "strategy", svc.AuthStrategyRef)

    // >>> END AUTHENTICATION <<<

    // Execute upstream request
    resp, err := h.client.Do(upstreamReq)
    if err != nil {
        return err
    }
    defer resp.Body.Close()

    // Proxy response
    return resp.Write(clientConn)
}
```
