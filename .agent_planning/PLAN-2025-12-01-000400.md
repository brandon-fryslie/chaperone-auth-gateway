2. If method denied, return 403 Forbidden with clear error message

**Pseudo-code**:
```go
// Lookup service
service, found := h.registry.Lookup(hostname)
if !found {
    // No service configured, allow (transparent proxy mode)
    return h.forwardToUpstream(ctx, clientConn, hostname, req)
}

// Check method
if err := h.enforcer.CheckMethod(req.Method, service.Policy); err != nil {
    h.logger.Warn("policy violation: method", "request_id", ctx.Value("request_id"), "method", req.Method, "error", err)
    return writeErrorResponse(clientConn, 403, "Forbidden: Method not allowed")
}

// Check path
if err := h.enforcer.CheckPath(req.URL.Path, service.Policy); err != nil {
    h.logger.Warn("policy violation: path", "request_id", ctx.Value("request_id"), "path", req.URL.Path, "error", err)
    return writeErrorResponse(clientConn, 403, "Forbidden: Path not allowed")
}

// Check body size
if req.ContentLength > 0 {
    if err := h.enforcer.CheckBodySize(req.ContentLength, service.Policy); err != nil {
        h.logger.Warn("policy violation: body size", "request_id", ctx.Value("request_id"), "size", req.ContentLength, "error", err)
        return writeErrorResponse(clientConn, 413, "Payload Too Large")
    }
}

// All policy checks passed, forward to upstream
return h.forwardToUpstream(ctx, clientConn, hostname, req)
```
