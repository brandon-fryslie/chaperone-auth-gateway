// internal/errors/errors.go - Created in Phase 0

package errors

import (
    "errors"
    "fmt"
    "net/http"
)

// Sentinel errors - use errors.Is() to check
var (
    ErrSecretNotFound    = errors.New("secret not found")
    ErrPermissionDenied  = errors.New("permission denied")
    ErrTimeout           = errors.New("operation timeout")
    ErrPolicyViolation   = errors.New("policy violation")
    ErrInvalidConfig     = errors.New("invalid configuration")
)

// Error types - use errors.As() to extract
type SecretError struct {
    Provider string
    Ref      string
    Cause    error
}

func (e *SecretError) Error() string {
    return fmt.Sprintf("secret error from %s[%s]: %v", e.Provider, e.Ref, e.Cause)
}

func (e *SecretError) Unwrap() error { return e.Cause }

type PolicyError struct {
    PolicyType string // "path", "method", "body_size"
    Value      string
    Cause      error
}

func (e *PolicyError) Error() string {
    return fmt.Sprintf("policy violation [%s]: %s", e.PolicyType, e.Value)
}

func (e *PolicyError) Unwrap() error { return e.Cause }

// HTTP error mapping
func HTTPStatus(err error) int {
    if errors.Is(err, ErrPolicyViolation) {
        return http.StatusForbidden
    }
    if errors.Is(err, ErrSecretNotFound) {
        return http.StatusServiceUnavailable
    }
    if errors.Is(err, ErrPermissionDenied) {
        return http.StatusForbidden
    }
    if errors.Is(err, ErrTimeout) {
        return http.StatusGatewayTimeout
    }

    var secretErr *SecretError
    if errors.As(err, &secretErr) {
        return http.StatusServiceUnavailable
    }

    var policyErr *PolicyError
    if errors.As(err, &policyErr) {
        return http.StatusForbidden
    }

    return http.StatusBadGateway // Default for unknown errors
}

// Safe error message for clients (redacts internals)
func ClientMessage(err error) string {
    if errors.Is(err, ErrPolicyViolation) {
        return "Request denied by policy"
    }
    if errors.Is(err, ErrSecretNotFound) {
        return "Service temporarily unavailable"
    }

    var policyErr *PolicyError
    if errors.As(err, &policyErr) {
        return fmt.Sprintf("Policy violation: %s not allowed", policyErr.PolicyType)
    }

    return "Internal proxy error" // Generic for security
}
```

**Impact of Current Plan:**
- Each phase will invent its own error handling (inconsistent patterns)
- Error information will leak to clients (security issue)
- Difficult to debug errors across component boundaries
- Refactoring error handling after implementation is painful

// internal/log/logger.go - Created in Phase 0

package log

import (
    "context"
    "log/slog"
    "os"
    "time"

    "github.com/google/uuid"
)

type contextKey string

const requestIDKey contextKey = "request_id"

// Initialize logger with JSON output and standard fields
var logger *slog.Logger

func init() {
    handler := slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
        Level: slog.LevelInfo,
        ReplaceAttr: func(groups []string, a slog.Attr) slog.Attr {
            // Redact sensitive headers
            if a.Key == "Authorization" || a.Key == "X-API-Key" {
                return slog.String(a.Key, "[REDACTED]")
            }
            return a
        },
    })
    logger = slog.New(handler)
}

// Create context with request ID
func WithRequestID(ctx context.Context) context.Context {
    reqID := uuid.New().String()
    return context.WithValue(ctx, requestIDKey, reqID)
}

// Get logger with request ID from context
func FromContext(ctx context.Context) *slog.Logger {
    reqID, ok := ctx.Value(requestIDKey).(string)
    if !ok {
        reqID = "unknown"
    }
    return logger.With("request_id", reqID)
}

// Standard log methods that enforce consistent fields
func Info(ctx context.Context, msg string, args ...any) {
    FromContext(ctx).Info(msg, args...)
}

func Error(ctx context.Context, msg string, err error, args ...any) {
    allArgs := append(args, "error", err.Error())
    FromContext(ctx).Error(msg, allArgs...)
}

func Debug(ctx context.Context, msg string, args ...any) {
    FromContext(ctx).Debug(msg, args...)
}

// Duration logging helper
func LogDuration(ctx context.Context, operation string, start time.Time) {
    duration := time.Since(start)
    Info(ctx, "operation completed",
        "operation", operation,
        "duration_ms", duration.Milliseconds())
}
```

**Correct Approach - Architecture Before Implementation:**

```go
// internal/shutdown/shutdown.go - Created in Phase 0

package shutdown

import (
    "context"
    "errors"
    "log/slog"
    "os"
    "os/signal"
    "sync"
    "syscall"
    "time"
)

// Manager coordinates graceful shutdown
type Manager struct {
    shutdownFuncs []func(ctx context.Context) error
    shutdownOnce  sync.Once
    logger        *slog.Logger
}

func NewManager(logger *slog.Logger) *Manager {
    return &Manager{
        shutdownFuncs: []func(ctx context.Context) error{},
        logger:        logger,
    }
}

// Register shutdown function (called in LIFO order)
func (m *Manager) Register(fn func(ctx context.Context) error) {
    m.shutdownFuncs = append(m.shutdownFuncs, fn)
}

// Wait for shutdown signal and execute shutdown sequence
func (m *Manager) WaitForShutdown() error {
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, os.Interrupt, syscall.SIGTERM)

    sig := <-sigChan
    m.logger.Info("shutdown signal received", "signal", sig.String())

    return m.Shutdown(30 * time.Second)
}

// Execute shutdown sequence with timeout
func (m *Manager) Shutdown(timeout time.Duration) error {
    var err error
    m.shutdownOnce.Do(func() {
        ctx, cancel := context.WithTimeout(context.Background(), timeout)
        defer cancel()

        // Execute shutdown functions in reverse order (LIFO)
        for i := len(m.shutdownFuncs) - 1; i >= 0; i-- {
            fn := m.shutdownFuncs[i]
            if fnErr := fn(ctx); fnErr != nil {
                m.logger.Error("shutdown function failed", "error", fnErr)
                err = errors.Join(err, fnErr)
            }
        }
    })

    return err
}
```

**Usage in Server:**
```go

    // Register shutdown handler

    })

    // Start server

// internal/config/config.go - Created in Phase 0

package config

import (
    "fmt"
    "os"

    "github.com/BurntSushi/toml"
)

type Config struct {
    Server   ServerConfig
    Services map[string]ServiceConfig
    Logging  LoggingConfig
}

type ServerConfig struct {
    Address string `toml:"address"`
    Port    int    `toml:"port"`
}

type ServiceConfig struct {
    HostPattern   string   `toml:"host_pattern"`
    AuthStrategy  string   `toml:"auth_strategy"`
    CredentialRef string   `toml:"credential_ref"`
    Policy        PolicyConfig `toml:"policy"`
}

type PolicyConfig struct {
    AllowedMethods []string `toml:"allowed_methods"`
    AllowedPaths   []string `toml:"allowed_paths"`
    MaxBodyBytes   int64    `toml:"max_body_bytes"`
}

type LoggingConfig struct {
    Level  string `toml:"level"`
    Format string `toml:"format"`
}

// Load config from file
func Load(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("read config: %w", err)
    }

    var cfg Config
    if err := toml.Unmarshal(data, &cfg); err != nil {
        return nil, fmt.Errorf("parse config: %w", err)
    }

    if err := cfg.Validate(); err != nil {
        return nil, fmt.Errorf("invalid config: %w", err)
    }

    return &cfg, nil
}

// Validate config
func (c *Config) Validate() error {
    if c.Server.Port < 1 || c.Server.Port > 65535 {
        return fmt.Errorf("invalid port: %d", c.Server.Port)
    }

    for name, svc := range c.Services {
        if svc.HostPattern == "" {
            return fmt.Errorf("service %s: host_pattern required", name)
        }
        if svc.AuthStrategy == "" {
            return fmt.Errorf("service %s: auth_strategy required", name)
        }
        if svc.CredentialRef == "" {
            return fmt.Errorf("service %s: credential_ref required", name)
        }
    }

    return nil
}
```

## Quality Assurance Requirements

### Test-First Development (MANDATORY)

Every implementation task MUST follow this sequence:

1. **Write failing tests** that define expected behavior
2. **Run tests** to verify they fail (red)
3. **Implement minimal code** to make tests pass
4. **Run tests** to verify they pass (green)
5. **Refactor** if needed
6. **Run tests again** to verify refactoring didn't break behavior
7. **Only then** move to next task

### Integration Testing Between Phases (MANDATORY)

After completing each phase:

1. **Write integration test** that exercises entire phase functionality
2. **Integration test must pass** before starting next phase
3. **Integration test becomes regression test** for future changes

Example:
- **Phase 1 integration test:** HTTPS tunnel to google.com works
- **Phase 2 integration test:** MITM tunnel to configured domain works
